	    MasterPackingSSRS.MtlJobNum AS JobNum, 
    MasterPackingSSRS.ProjectID, 
	    MasterPackingSSRS.ProjDesc As ProjName, 
     MAX(JobHead.ReqDueDate) AS ReqDueDate,

        WHEN MasterPackingSSRS.JobClosed = 1 THEN ISNULL(ShipDate, MasterPackingSSRS.ClosedDate)

    END) AS ShipDate

    AND MasterPackingSSRS.JobNum = JobHead.JobNum

GROUP BY 
    MasterPackingSSRS.OrderNum, 
    MasterPackingSSRS.MtlJobNum, 
    MasterPackingSSRS.ProjectID, 
	MasterPackingSSRS.ProjDesc,

    MasterPackingSSRS.ClosedDate

    MONTH(ShipDate) AS ShipMonth,
    DATENAME(month, ShipDate) AS MonthName,
    COUNT(JobNum) AS TotalShipments

WHERE YEAR(ShipDate) = YEAR(GETDATE()) - 1
